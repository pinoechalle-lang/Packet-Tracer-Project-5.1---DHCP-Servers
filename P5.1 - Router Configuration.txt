=========================================================================================
Router-A
=========================================================================================

=============[ Escalate Mode ]=============

enable
configure terminal


=============[ Basic Configurations ]=============

hostname Router-A

no ip domain


========[ Enabling and Assigning IP Address to each Interface ]========

interface g0/1
 description Link to Switch A - Utilized as Sub-Interfaces for VLAN 2,3
 no shutdown
exit

interface g0/0
 description Link to Switch C - Utilized as Sub-Interface for VLAN 900 (DHCP Server)
 no shutdown
exit

interface s0/0/0
 description Link to Router B
 ip address 10.0.2.9 255.255.255.248
 no shutdown
exit


========[ Enabling and Assigning IP Address to each Sub-Interface for VLANs ]========

interface g0/1.2
 description Sub-Interface for VLAN 2
 encapsulation dot1q 2
 ip address 192.168.2.254 255.255.255.0
 ip helper-address 10.0.2.17
exit


interface g0/1.3
 description Sub-Interface for VLAN 3
 encapsulation dot1q 3
 ip address 192.168.3.254 255.255.255.0
 ip helper-address 10.0.2.17
exit


// This interface acts as a DHCP client sharing the same subnet as the DHCP server:

interface g0/0.900
 description Sub-Interface for VLAN 900 (DHCP Server)
 encapsulation dot1q 900
 ip address dhcp
exit


=============[ IP Route ]=============

ip route 192.168.4.0 255.255.255.0 10.0.2.14
 
ip route 192.168.5.0 255.255.255.0 10.0.2.14

ip route 172.19.20.0 255.255.255.0 10.0.2.14


=============[ Save Running Configuration (Global) ]=============

do copy running-config startup-config



=========================================================================================
Router-B:
=========================================================================================

=============[ Escalate Mode ]=============

enable
configure terminal


=============[ Basic Configurations ]=============

hostname Router-B

no ip domain


========[ Enabling and Assigning IP Address to each Interface ]========

interface g0/1
 description Link to Switch B - Utilized as Sub-Interfaces for VLAN 4,5
 no shutdown
exit

interface g0/0
 description Link to DNS Server
 ip address 172.19.20.254 255.255.255.0
 no shutdown
exit

interface s0/0/0
 description Link to Router A
 ip address 10.0.2.14 255.255.255.248
 no shutdown
exit


========[ Enabling and Assigning IP Address to each Sub-Interface for VLANs ]========

interface g0/1.4
 description Sub-Interface for VLAN 4
 encapsulation dot1q 4
 ip address 192.168.4.254 255.255.255.0
exit

interface g0/1.5
 description Sub-Interface for VLAN 5
 encapsulation dot1q 5
 ip address 192.168.5.254 255.255.255.0
exit


=============[ Configuring DHCP Pools (Groups) and Excluded Addresses ]=============

ip dhcp excluded-address 192.168.4.1 192.168.4.4
ip dhcp excluded-address 192.168.4.254

ip dhcp excluded-address 192.168.5.1 192.168.5.4
ip dhcp excluded-address 192.168.5.254


ip dhcp pool VLAN-4
 network 192.168.4.0 255.255.255.0
 default-router 192.168.4.254
 dns-server 172.19.20.1
 domain-name companyrecords.local
exit

ip dhcp pool VLAN-5
 network 192.168.5.0 255.255.255.0
 default-router 192.168.5.254
 dns-server 172.19.20.1
 domain-name companyrecords.local
exit


=============[ IP Route ]=============

ip route 192.168.2.0 255.255.255.0 10.0.2.9

ip route 192.168.3.0 255.255.255.0 10.0.2.9

ip route 10.0.2.16 255.255.255.248 10.0.2.9


=============[ Save Running Configuration (Global) ]=============

do copy running-config startup-config
